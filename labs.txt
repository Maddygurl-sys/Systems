const express = require('express');
const app = express();


app.use(express.json());


let patients = [
    { id: 1, name: 'Arun', age: 25, disease: 'Fever' },
    { id: 2, name: 'Priya', age: 30, disease: 'Cold' }
];


// Get all patients
app.get('/api/patients', (req, res) => {
    res.json(patients);
});


// Get one patient
app.get('/api/patients/:id', (req, res) => {
    const patient = patients.find(p => p.id === Number(req.params.id));


    if (!patient) {
        return res.status(404).send('Patient not found');
    }


    res.json(patient);
});


// Add a patient
app.post('/api/patients', (req, res) => {
    const patient = {
        id: patients.length
            ? Math.max(...patients.map(p => p.id)) + 1 : 1,
        name: req.body.name,
        age: req.body.age,
        disease: req.body.disease
    };


    patients.push(patient);
    res.status(201).json(patient);
});


// Update a patient
app.put('/api/patients/:id', (req, res) => {
    const patient = patients.find(p => p.id === Number(req.params.id));


    if (!patient) {
        return res.status(404).send('Patient not found');
    }


    patient.name = req.body.name ?? patient.name;
    patient.age = req.body.age ?? patient.age;
    patient.disease = req.body.disease ?? patient.disease;


    res.json(patient);
});


// Delete a patient
app.delete('/api/patients/:id', (req, res) => {
    const index = patients.findIndex(
        p => p.id === Number(req.params.id)
    );


    if (index === -1) {
        return res.status(404).send('Patient not found');
    }


    patients.splice(index, 1);
    res.send('Patient deleted successfully');
});


app.listen(8080, () => {
    console.log('Hospital API running on port 8080');
});

Hadoop:


1. Install Java and SSH
Run inside Ubuntu:
sudo apt update
sudo apt install -y openjdk-11-jdk ssh
java -version
2. Download and extract Hadoop
cd ~
wget https://dlcdn.apache.org/hadoop/common/hadoop-3.3.6/hadoop-3.3.6.tar.gz
tar -xzf hadoop-3.3.6.tar.gz
mv hadoop-3.3.6 hadoop
If Hadoop is already installed, skip the download and rename commands.
3. Set environment variables
Find your Java path:
dirname $(dirname $(readlink -f $(which java)))
Open the environment file:
gedit ~/.bashrc
Add the following lines, using the actual Java path returned above:
export JAVA_HOME=/usr/lib/jvm/java-11-openjdk-amd64
export HADOOP_HOME=HADOOP_HOME
export HADOOP_MAPRED_HOME=HADOOP_HOME
export HADOOP_HDFS_HOME=HADOOP_HOME
export PATH=HADOOP_HOME/bin:$HADOOP_HOME/sbin
Save the file, then run:
source ~/.bashrc
4. Configure Hadoop files
Go to the configuration folder:
cd ~/hadoop/etc/hadoop
A. hadoop-env.sh
gedit hadoop-env.sh
Set the Java path:
export JAVA_HOME=/usr/lib/jvm/java-11-openjdk-amd64
Save the file.
B. core-site.xml
gedit core-site.xml
Add inside the existing <configuration> tags:
<property>
    <name>fs.defaultFS</name>
    <value>hdfs://localhost:9000</value>
</property>
C. hdfs-site.xml
gedit hdfs-site.xml
Add inside <configuration>:
<property>
    <name>dfs.replication</name>
    <value>3</value>
</property>
<property>
    <name>dfs.name.dir</name>
    <value>/home/YOUR_USERNAME/hadoop/data/namenode</value>
</property>
<property>
    <name>dfs.data.dir</name>
    <value>/home/YOUR_USERNAME/hadoop/data/datanode</value>
</property>
Replace YOUR_USERNAME with your Ubuntu username.
D. mapred-site.xml
gedit mapred-site.xml
<property>
    <name>mapreduce.framework.name</name>
    <value>yarn</value>
</property>
E. yarn-site.xml
gedit yarn-site.xml
<property>
    <name>yarn.nodemanager.aux-services</name>
    <value>mapreduce_shuffle</value>
</property>
<property>
    <name>yarn.nodemanager.auxservices.mapreduce.shuffle.class</name>
    <value>org.apache.hadoop.mapred.ShuffleHandler</value>
</property>
For each XML file, preserve its existing <configuration> and </configuration> tags. Save the changes.
5. Create the data directories
cd ~/hadoop
mkdir -p data/namenode data/datanode
6. Configure SSH
ssh-keygen -t rsa
Press Enter through the prompts to accept the defaults.
cat ~/.ssh/id_rsa.pub >> ~/.ssh/authorized_keys
chmod 640 ~/.ssh/authorized_keys
7. Format the NameNode
Run this only for a fresh Hadoop installation.
hdfs namenode -format
Do not repeat this on an existing HDFS setup containing data.
8. Start Hadoop
Start HDFS:
start-dfs.sh
Check the processes:
jps
Start YARN:
start-yarn.sh
Check again:
jps
Expected services:
* NameNode
* DataNode
* SecondaryNameNode
* ResourceManager
* NodeManager
9. Open the web interfaces
* HDFS: http://localhost:9870
* YARN: http://localhost:8088
10. Stop Hadoop
stop-yarn.sh
stop-dfs.sh

package org.cloudbus.cloudsim.examples;


import org.cloudbus.cloudsim.*;
import org.cloudbus.cloudsim.core.CloudSim;
import java.util.*;


public class BasicResourceAllocation {


    public static void main(String[] args) throws Exception {


        CloudSim.init(1, Calendar.getInstance(), false);


        // Create Host
        List<Pe> peList = new ArrayList<>();
        peList.add(new Pe(0, new PeProvisionerSimple(1000)));


        List<Host> hostList = new ArrayList<>();
        hostList.add(new Host(0,
                new RamProvisionerSimple(2048),
                new BwProvisionerSimple(10000),
                1000000, peList,
                new VmSchedulerTimeShared(peList)));


        // Create Datacenter
        DatacenterCharacteristics dc = new DatacenterCharacteristics(
                "x86", "Linux", "Xen", hostList,
                10.0, 3.0, 0.05, 0.001, 0.0);


        new Datacenter("Datacenter_0", dc,
                new VmAllocationPolicySimple(hostList),
                new LinkedList<Storage>(), 0);


        // Create Broker
        DatacenterBroker broker = new DatacenterBroker("Broker");
        int brokerId = broker.getId();


        // Create 2 VMs
        List<Vm> vmList = new ArrayList<>();


        for (int i = 0; i < 2; i++) {
            vmList.add(new Vm(i, brokerId, 500, 1,
                    512, 1000, 10000, "Xen",
                    new CloudletSchedulerTimeShared()));
        }


        broker.submitVmList(vmList);


        // Create 2 Cloudlets
        List<Cloudlet> cloudletList = new ArrayList<>();
        UtilizationModel model = new UtilizationModelFull();


        for (int i = 0; i < 2; i++) {
            Cloudlet c = new Cloudlet(i, 10000, 1,
                    300, 300, model, model, model);
            c.setUserId(brokerId);
            cloudletList.add(c);
        }


        broker.submitCloudletList(cloudletList);


        // Run Simulation
        CloudSim.startSimulation();
        CloudSim.stopSimulation();


        // Print Results
        for (Cloudlet c : broker.getCloudletReceivedList()) {
            System.out.println("Cloudlet " + c.getCloudletId()
                    + " | Status: "
                    + (c.getStatus() == Cloudlet.SUCCESS
                    ? "SUCCESS" : "FAILED")
                    + " | VM: " + c.getVmId());
        }
    }
}
